class Solution:
    def findColoredRoot(self, edges: list[list[int]], colors: str) -> int:
        next_color = {"R": "B", "B": "W", "W": "R"}
        in_degree = [0] * len(colors)
        out_degree = [0] * len(colors)

        for u, v in edges:
            if next_color[colors[u]] == colors[v]:
                parent, child = u, v
            elif next_color[colors[v]] == colors[u]:
                parent, child = v, u
            else:  # same color on both ends
                return -1

            in_degree[child] += 1
            out_degree[parent] += 1
            if in_degree[child] > 1 or out_degree[parent] > 2:
                return -1

        return in_degree.index(0)
```

#include <cctype>
#include <cstddef>
#include <string>

class Solution {
public:
    bool isPalindrome(std::string s) {
        std::size_t left = 0;
        std::size_t right = s.size() - 1;
        while (left < right) {
            if (!std::isalnum(s[left])) {
                ++left;
                continue;
            }
            if (!std::isalnum(s[right])) {
                --right;
                continue;
            }

            if (std::tolower(s[left]) != std::tolower(s[right])) {
                return false;
            }
            ++left;
            --right;
        }
        return true;
    }
};
```

```cpp
class Solution {
public:
    bool isPalindrome(int x) {
        if (x < 0) { return false; }

        int num_digits = 0;
        int x_copy = x;
        while (0 < x_copy) {
            x_copy /= 10;
            ++num_digits;
        }

        int reversed_last = 0;
        for (int i = 0; i < num_digits / 2; ++i) {
            reversed_last = reversed_last * 10 + (x % 10);
            x /= 10;
        }

        if (num_digits % 2 != 0) {
            x /= 10;
        }
        return reversed_last == x;
    }
};
```

```cpp
class Solution {
public:
    bool isPalindrome(int x) {
        if (x < 0) { return false; }
        // Without this, leading zeros will be discarded incorrectly
        // in the following operations
        if (x != 0 && x % 10 == 0) { return false; }

        int reversed_last = 0;
        while (reversed_last < x) {
            reversed_last = reversed_last * 10 + (x % 10);
            x /= 10;
        }
        return reversed_last == x || reversed_last / 10 == x;
    }
};
```

#include <algorithm>

class Solution {
public:
    int missingNumber(std::vector<int>& nums) {
        std::sort(nums.begin(), nums.end());
        for (int i = 0; i < nums.size(); ++i) {
            if (nums[i] != i) { return i; }
        }
        return static_cast<int>(nums.size());
    }
};

#include <vector>

class Solution {
public:
    int missingNumber(std::vector<int>& nums) {
        int total = std::accumulate(nums.begin(), nums.end(), 0);
        int n = static_cast<int>(nums.size());
        int expected = (n * (n + 1)) / 2;
        return expected - total;
    }
};

        int i = static_cast<int>(s.size() - 1);
        while (0 <= i && s[i] == ' ') { --i; }
        int length = 0;
        while (0 <= i && s[i] != ' ') {

            ++length;
        }
