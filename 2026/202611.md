```py
import math


class Solution:
    def updateMatrix(self, mat: list[list[int]]) -> list[list[int]]:
        num_rows = len(mat)
        num_cols = len(mat[0])
        dist = [[float("inf")] * num_cols for _ in range(num_rows)]

        cells = []
        for row in range(num_rows):
            for col in range(num_cols):
                if mat[row][col] == 0:
                    cells.append((row, col))

        distance = 0
        while cells:
            next_cells = []
            for row, col in cells:
                if not (0 <= row < num_rows and 0 <= col < num_cols):
                    continue
                if not math.isinf(dist[row][col]):
                    continue

                dist[row][col] = distance
                next_cells.append((row + 1, col))
                next_cells.append((row - 1, col))
                next_cells.append((row, col + 1))
                next_cells.append((row, col - 1))

            distance += 1
            cells = next_cells

        return dist
```

```py
class Solution:
    def maxAreaOfIsland(self, grid: list[list[int]]) -> int:
        num_rows = len(grid)
        num_cols = len(grid[0])

        visited = [[False] * num_cols for _ in range(num_rows)]
        def count_area(row: int, col: int) -> int:
            if not (0 <= row < num_rows and 0 <= col < num_cols):
                return 0
            if grid[row][col] == 0:
                return 0
            if visited[row][col]:
                return 0

            visited[row][col] = True
            count = 1
            count += count_area(row - 1, col)
            count += count_area(row + 1, col)
            count += count_area(row, col - 1)
            count += count_area(row, col + 1)
            return count

        max_area = 0
        for row in range(num_rows):
            for col in range(num_cols):
                max_area = max(max_area, count_area(row, col))

        return max_area
```

```py
class Solution:
    def insert(self, intervals: list[list[int]], newInterval: list[int]) -> list[list[int]]:
        merged = []
        merged_start, merged_end = newInterval

        i = 0
        while i < len(intervals) and intervals[i][1] < merged_start:
            merged.append(intervals[i].copy())
            i += 1

        while i < len(intervals) and intervals[i][0] <= merged_end:
            merged_end = max(merged_end, intervals[i][1])
            merged_start = min(merged_start, intervals[i][0])
            i += 1
        merged.append([merged_start, merged_end])

        while i < len(intervals):
            merged.append(intervals[i].copy())
            i += 1

        return merged
```

```py
class Solution:
    def maxProfit(self, prices: list[int]) -> int:
        suffix_min = [float("inf")] * (len(prices) + 1)
        for i in range(1, len(prices) + 1):
            suffix_min[i] = min(suffix_min[i - 1], prices[i - 1])

        max_profit = [0] * (len(prices) + 1)
        for i in range(len(prices)):
            max_profit[i + 1] = max(
                max_profit[i],
                prices[i] - suffix_min[i + 1],
            )

        return max_profit[-1]
```

```py
class Solution:
    def maxSubArray(self, nums: list[int]) -> int:
        prefix_sum = [0] * (len(nums) + 1)
        for i in range(1, len(nums) + 1):
            prefix_sum[i] = prefix_sum[i - 1] + nums[i - 1]

        max_sum = max(nums)
        min_prefix = float("inf")
        for i in range(len(prefix_sum)):
            if prefix_sum[i] < min_prefix:
                min_prefix = prefix_sum[i]
                continue

            max_sum = max(max_sum, prefix_sum[i] - min_prefix)

        return max_sum
```

```py
        count = 1
        candidate = float("inf")
        for num in nums:
            if num == candidate:
                count += 1
                continue

            count -= 1
            if count == 0:
```

```py
class Solution:
    def addBinary(self, a: str, b: str) -> str:
        len_longer = max(len(a), len(b))
        a = a.rjust(len_longer, "0")
        b = b.rjust(len_longer, "0")
        carry = 0
        result = []
        for i in range(len_longer):
            total = int(a[-(i + 1)]) + int(b[-(i + 1)]) + carry
            result.append(str(total % 2))
            carry = total // 2

        if 0 < carry:
            result.append("1")

        return "".join(result[::-1])
```

```py
    def middleNode(self, head: ListNode | None) -> ListNode | None:
        if head is None:
            return None

        node = head
        count = 0
        while node is not None:
            count += 1
            node = node.next

        node = head
        for i in range(count // 2):
            node = node.next

        return node
```

```py
class Solution:
    def reverseList(self, head: ListNode | None) -> ListNode | None:
        reversing = head
        reversed_head = None
        while reversing is not None:
            next_reversing = reversing.next
            reversing.next = reversed_head
            reversed_head = reversing
            reversing = next_reversing

        return reversed_head
```

```py
class Solution:
    def reverseList(self, head: ListNode | None) -> ListNode | None:
        def reverse(node: ListNode) -> tuple[ListNode, ListNode]:
            if node.next is None:
                return node, node

            reversed_head, reversed_tail = reverse(node.next)
            reversed_tail.next = node
            node.next = None

            return reversed_head, node

        if head is None:
            return None

        return reverse(head)[0]
```

```py
import collections


class Solution:
    def longestPalindrome(self, s: str) -> int:
        char_to_count = collections.Counter(s)
        max_length = 0
        for count in char_to_count.values():
            if 2 <= count:
                if count % 2 == 0:
                    max_length += count
                else:
                    max_length += count - 1

        if max_length < len(s):
            max_length += 1

        return max_length
```
