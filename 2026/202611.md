```py
import collections


class Solution:
    def majorityElement(self, nums: list[int]) -> int:
        num_to_count = collections.Counter(nums)
        for num, count in num_to_count.items():
            if len(nums) // 2 < count:
                return num

        raise ValueError("no majority element found.")
```

```py
class Solution:
    def addBinary(self, a: str, b: str) -> str:
        len_longer = max(len(a), len(b))
        a = a.rjust(len_longer, "0")
        b = b.rjust(len_longer, "0")
        carry = 0
        result = []
        for i in range(len_longer):
            total = int(a[-(i + 1)]) + int(b[-(i + 1)]) + carry
            result.append(str(total % 2))
            carry = total // 2

        if 0 < carry:
            result.append("1")

        return "".join(result[::-1])
```

```py
    def middleNode(self, head: ListNode | None) -> ListNode | None:
        if head is None:
            return None

        slow = head
        fast = head
        while fast is not None and fast.next is not None:
            slow = slow.next
            fast = fast.next.next

        return slow
```

```py
class Solution:
    def middleNode(self, head: ListNode | None) -> ListNode | None:
        if head is None:
            return None

        node = head
        count = 0
        while node is not None:
            count += 1
            node = node.next

        node = head
        for i in range(count // 2):
            node = node.next

        return node
```

```py
# Definition for singly-linked list.

#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
class Solution:
    def reverseList(self, head: ListNode | None) -> ListNode | None:
        reversing = head
        reversed_head = None
        while reversing is not None:
            next_reversing = reversing.next
            reversing.next = reversed_head
            reversed_head = reversing
            reversing = next_reversing

        return reversed_head
```

```py
class Solution:
    def reverseList(self, head: ListNode | None) -> ListNode | None:
        def reverse(node: ListNode) -> tuple[ListNode, ListNode]:
            if node.next is None:
                return node, node

            reversed_head, reversed_tail = reverse(node.next)
            reversed_tail.next = node
            node.next = None

            return reversed_head, node

        if head is None:
            return None

        return reverse(head)[0]
```

```py
import collections


class Solution:
    def longestPalindrome(self, s: str) -> int:
        char_to_count = collections.Counter(s)
        max_length = 0
        for count in char_to_count.values():
            if 2 <= count:
                if count % 2 == 0:
                    max_length += count
                else:
                    max_length += count - 1

        if max_length < len(s):
            max_length += 1

        return max_length
```
