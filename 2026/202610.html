class Solution {
public:
    bool isPowerOfTwo(int n) {
        if (n <= 0) { return false; }

        while (1 < n) {
            if (n % 2 == 1) { return false; }
            n = n / 2;  // n /= 2;
        }

    }

class Solution {
public:
    bool isPowerOfFour(int n) {
        const int MASK = 0x55555555;

        if (n <= 0) { return false; }
        if ((n & MASK) != n) { return false; }
        return (n & (n - 1)) == 0;
    }
};
