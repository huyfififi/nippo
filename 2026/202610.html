class Solution:
    def calculateScore(self, a: int, b: int) -> int:

class Solution:
    def maximizeSum(self, nums: list[int], k: int) -> int:
        improvements = []

        for num in nums:
            max_num = int("".join(sorted(str(num), reverse=True)))
            improvements.append(max_num - num)

        best_improvements = heapq.nlargest(k, improvements)
        return sum(nums) + sum(best_improvements)

    • The CPU only ever executes machine code. Assembly is just a human-readable text form of machine code. Compiled languages like C/C++/Rust are translated ahead of time into machine code for the target CPU (ARM64 on a MacBook, ARMv8-M Thumb-2 on the Pico 2’s Cortex-M33). Python (CPython or MicroPython) is not turned into machine code: it is compiled to bytecode, which an interpreter (itself a C program compiled to machine code) reads and executes. That’s why Python is slower. JIT runtimes like PyPy, Java and V8 generate machine code at runtime for frequently used code.

class Solution:
    def findColoredRoot(self, edges: list[list[int]], colors: str) -> int:
        next_color = {"R": "B", "B": "W", "W": "R"}
        in_degree = [0] * len(colors)
        out_degree = [0] * len(colors)

        for u, v in edges:
            if next_color[colors[u]] == colors[v]:
                parent, child = u, v
            elif next_color[colors[v]] == colors[u]:
                parent, child = v, u
            else:  # same color on both ends
                return -1

            in_degree[child] += 1
            out_degree[parent] += 1
            if in_degree[child] > 1 or out_degree[parent] > 2:
                return -1

        return in_degree.index(0)

#include <cctype>

class Solution {
public:
    bool isPalindrome(std::string s) {
        std::size_t left = 0;
        std::size_t right = s.size() - 1;
        while (left < right) {
            if (!std::isalnum(s[left])) {
                ++left;
                continue;
            }
            if (!std::isalnum(s[right])) {
                --right;
                continue;
            }

            if (std::tolower(s[left]) != std::tolower(s[right])) {
                return false;
            }
            ++left;
            --right;
        }
        return true;
    }
};

class Solution {
public:
    bool isPalindrome(int x) {
        if (x < 0) { return false; }

        int num_digits = 0;
        int x_copy = x;
        while (0 < x_copy) {
            x_copy /= 10;
            ++num_digits;
        }

        int reversed_last = 0;
        for (int i = 0; i < num_digits / 2; ++i) {
            reversed_last = reversed_last * 10 + (x % 10);
            x /= 10;
        }

        if (num_digits % 2 != 0) {
            x /= 10;
        }
        return reversed_last == x;
    }
};
class Solution {
public:
    bool isPalindrome(int x) {
        if (x < 0) { return false; }
        // Without this, leading zeros will be discarded incorrectly
        // in the following operations
        if (x != 0 && x % 10 == 0) { return false; }

        int reversed_last = 0;
        while (reversed_last < x) {
            reversed_last = reversed_last * 10 + (x % 10);
            x /= 10;
        }
        return reversed_last == x || reversed_last / 10 == x;
    }
};

    }
};
