#include <string>

class Solution {
public:
    int lengthOfLastWord(std::string s) {
        int i = static_cast<int>(s.size() - 1);
        while (0 <= i && s[i] == ' ') { --i; }
        int length = 0;
        while (0 <= i && s[i] != ' ') {
            --i;
            ++length;
        }
        return length;
    }
};

class Solution {
public:
    bool isPowerOfTwo(int n) {
        if (n <= 0) { return false; }

        while (1 < n) {
            if (n % 2 == 1) { return false; }
            n = n / 2;  // n /= 2;
        }

        return true;
    }
};

class Solution {
public:
    bool isPowerOfFour(int n) {
        const int MASK = 0x55555555;

        if (n <= 0) { return false; }
        if ((n & MASK) != n) { return false; }
        return (n & (n - 1)) == 0;
    }
};
